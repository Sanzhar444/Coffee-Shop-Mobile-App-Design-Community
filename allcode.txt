
import React, { useState } from 'react';
import { View, StyleSheet, Alert, SafeAreaView, StatusBar } from 'react-native';

import { COLORS } from './src/theme';
import { CoffeeProduct } from './src/data';
import Header from './src/components/Header';
import BottomNav from './src/components/BottomNav';
import Login from './src/components/Login';

import Home from './src/screens/Home';
import Catalog from './src/screens/Catalog';
import Cart, { CartItem } from './src/screens/Cart';
import Profile from './src/screens/Profile';
import Details from './src/screens/Details';

export default function App() {
  const [activeTab, setActiveTab] = useState('Главная');
  const [cart, setCart] = useState<CartItem[]>([]);
  const [selectedProduct, setSelectedProduct] = useState<CoffeeProduct | null>(null);
  const [showLogin, setShowLogin] = useState(false);
  const [email, setEmail] = useState('');

  const addToCart = (product: CoffeeProduct) => {
    setCart((old) => {
      const found = old.find((item) => item.product.id === product.id);

      if (found) {
        return old.map((item) =>
          item.product.id === product.id
            ? { ...item, quantity: item.quantity + 1 }
            : item
        );
      }

      return [...old, { product, quantity: 1 }];
    });

    Alert.alert('Добавлено', `${product.name} добавлен в корзину`);
  };

  const changeQuantity = (id: number, amount: number) => {
    setCart((old) =>
      old
        .map((item) =>
          item.product.id === id
            ? { ...item, quantity: item.quantity + amount }
            : item
        )
        .filter((item) => item.quantity > 0)
    );
  };

  const checkout = () => {
    if (!cart.length) {
      Alert.alert('Корзина пуста');
      return;
    }

    Alert.alert('Заказ принят!', 'Спасибо за ваш заказ.');
    setCart([]);
    setActiveTab('Главная');
  };

  const cartCount = cart.reduce((sum, item) => sum + item.quantity, 0);

  const renderScreen = () => {
    if (showLogin) {
      return (
        <Login
          onBack={() => setShowLogin(false)}
          onLogin={(userEmail) => {
            setEmail(userEmail);
            setShowLogin(false);
            setActiveTab('Профиль');
          }}
        />
      );
    }

    if (selectedProduct) {
      return (
        <Details
          product={selectedProduct}
          onBack={() => setSelectedProduct(null)}
          onAdd={() => addToCart(selectedProduct)}
        />
      );
    }

    switch (activeTab) {
      case 'Каталог':
        return (
          <Catalog
            onAdd={addToCart}
            onSelect={setSelectedProduct}
          />
        );

      case 'Корзина':
        return (
          <Cart
            cart={cart}
            onChange={changeQuantity}
            onCheckout={checkout}
          />
        );

      case 'Профиль':
        return (
          <Profile
            email={email}
            onLogin={() => setShowLogin(true)}
            onLogout={() => setEmail('')}
          />
        );

      default:
        return (
          <Home
            onAdd={addToCart}
            onSelect={setSelectedProduct}
            onCatalog={() => setActiveTab('Каталог')}
          />
        );
    }
  };

  return (
    <SafeAreaView style={styles.safe}>
      <StatusBar barStyle="dark-content" />

      {!showLogin && !selectedProduct && (
        <Header
          cartCount={cartCount}
          onCartPress={() => setActiveTab('Корзина')}
        />
      )}

      <View style={styles.content}>
        {renderScreen()}
      </View>

      {!showLogin && !selectedProduct && (
        <BottomNav
          activeTab={activeTab}
          onChange={setActiveTab}
          cartCount={cartCount}
        />
      )}
    </SafeAreaView>
  );
}

const styles = StyleSheet.create({
  safe: {
    flex: 1,
    backgroundColor: COLORS.background,
  },
  content: {
    flex: 1,
    paddingHorizontal: 18,
  },
});

export default function Catalog({ onAdd, onSelect }: Props) {
  const [search, setSearch] = useState('');
  const [category, setCategory] = useState('Все');

  const categories = ['Все', 'Кофе', 'Десерты', 'Холодный кофе'];

  const filtered = products.filter((p) =>
    (category === 'Все' || p.category === category) &&
    p.name.toLowerCase().includes(search.toLowerCase())
  );

  return (
    <ScrollView style={styles.container}>
      <Text style={styles.title}>Наше меню</Text>
      <Text style={styles.subtitle}>Найдите свой любимый вкус</Text>

      <TextInput
        style={styles.search}
        placeholder="Поиск кофе и десертов..."
        placeholderTextColor={COLORS.gray}
        value={search}
        onChangeText={setSearch}
      />

      <ScrollView horizontal showsHorizontalScrollIndicator={false}>
        {categories.map((item) => (
          <Text
            key={item}
            onPress={() => setCategory(item)}
            style={[
              styles.category,
              category === item && styles.activeCategory,
            ]}
          >
            {item}
          </Text>
        ))}
      </ScrollView>

      <View style={styles.grid}>
        {filtered.map((product) => (
          <ProductCard
            key={product.id}
            product={product}
            onPress={() => onSelect(product)}
            onAdd={() => onAdd(product)}
          />
        ))}
      </View>

      {filtered.length === 0 && (
        <Text style={styles.empty}>Товары не найдены</Text>
      )}
    </ScrollView>
  );
}

import React from 'react';

export type CartItem = {
  product: CoffeeProduct;
  quantity: number;
};

type Props = {
  cart: CartItem[];
  onChange: (id: number, amount: number) => void;
  onCheckout: () => void;
};

export default function Cart({ cart, onChange, onCheckout }: Props) {
  const total = cart.reduce(
    (sum, item) => sum + item.product.price * item.quantity,
    0
  );

  return (
    <ScrollView style={styles.container}>
      <Text style={styles.title}>Корзина</Text>
      <Text style={styles.subtitle}>Ваш заказ</Text>

      {cart.length === 0 ? (
        <View style={styles.empty}>
          <Text style={styles.emptyIcon}>🛍</Text>
          <Text style={styles.emptyTitle}>Пока здесь пусто</Text>
          <Text style={styles.emptyText}>
            Добавьте что-нибудь вкусное из каталога.
          </Text>
        </View>
      ) : (
        <>
          {cart.map((item) => (
            <View style={styles.item} key={item.product.id}>
              <Text style={styles.emoji}>☕</Text>
              <View style={styles.itemInfo}>
                <Text style={styles.name}>{item.product.name}</Text>
                <Text style={styles.price}>
                  {item.product.price.toLocaleString('ru-RU')} ₸
                </Text>
                <View style={styles.controls}>
                  <TouchableOpacity
                    style={styles.control}
                    onPress={() => onChange(item.product.id, -1)}
                  >
                    <Text>−</Text>
                  </TouchableOpacity>
                  <Text style={styles.quantity}>{item.quantity}</Text>
                  <TouchableOpacity
                    style={styles.control}
                    onPress={() => onChange(item.product.id, 1)}
                  >
                    <Text>+</Text>
                  </TouchableOpacity>
                </View>
              </View>
            </View>
          ))}

          <View style={styles.summary}>
            <Text style={styles.totalLabel}>Итого</Text>
            <Text style={styles.total}>
              {total.toLocaleString('ru-RU')} ₸
            </Text>
            <TouchableOpacity style={styles.checkout} onPress={onCheckout}>
              <Text style={styles.checkoutText}>Оформить заказ</Text>
            </TouchableOpacity>
          </View>
        </>
      )}
    </ScrollView>
  );
}

import React, { useState } from 'react';
import {
  View,
  Text,
  TextInput,
  TouchableOpacity,
  StyleSheet,
  ScrollView,
  Alert,
} from 'react-native';

type LoginProps = {
  onBack: () => void;
  onLogin: (email: string, password: string) => void;
};

const COLORS = {
  background: '#F5EBE0',
  card: '#FFF9F2',
  brown: '#4A2C20',
  dark: '#2C1D11',
  beige: '#D4A373',
  lightBrown: '#A98467',
  text: '#35251C',
  gray: '#8B8179',
  white: '#FFFFFF',
  border: '#E8D8C8',
};

export default function Login({ onBack, onLogin }: LoginProps) {
  const [screen, setScreen] = useState<'login' | 'register' | 'forgot'>(
    'login'
  );

  const [name, setName] = useState('');
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [confirmPassword, setConfirmPassword] = useState('');

  const [showPassword, setShowPassword] = useState(false);
  const [showConfirmPassword, setShowConfirmPassword] = useState(false);

  const validateEmail = (value: string) => {
    return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(value);
  };

  const handleLogin = () => {
    if (!email.trim() || !password.trim()) {
      Alert.alert('Ошибка', 'Заполните все поля.');
      return;
    }

    if (!validateEmail(email.trim())) {
      Alert.alert('Ошибка', 'Введите правильный email.');
      return;
    }

    if (password.length < 6) {
      Alert.alert(
        'Ошибка',
        'Пароль должен содержать минимум 6 символов.'
      );
      return;
    }

    onLogin(email.trim(), password);
  };

  const handleRegister = () => {
    if (!name.trim() || !email.trim() || !password || !confirmPassword) {
      Alert.alert('Ошибка', 'Заполните все поля.');
      return;
    }

    if (!validateEmail(email.trim())) {
      Alert.alert('Ошибка', 'Введите правильный email.');
      return;
    }

    if (password.length < 6) {
      Alert.alert(
        'Ошибка',
        'Пароль должен содержать минимум 6 символов.'
      );
      return;
    }

    if (password !== confirmPassword) {
      Alert.alert('Ошибка', 'Пароли не совпадают.');
      return;
    }

    Alert.alert(
      'Регистрация успешна!',
      'Ваш аккаунт создан.',
      [
        {
          text: 'Продолжить',
          onPress: () => onLogin(email.trim(), password),
        },
      ]
    );
  };

  const handleForgot = () => {
    if (!email.trim()) {
      Alert.alert('Ошибка', 'Введите ваш email.');
      return;
    }

    if (!validateEmail(email.trim())) {
      Alert.alert('Ошибка', 'Введите правильный email.');
      return;
    }

    Alert.alert(
      'Готово!',
      'Если аккаунт существует, инструкция по восстановлению будет отправлена на вашу почту.'
    );

    setScreen('login');
  };

  const renderInput = (
    label: string,
    placeholder: string,
    value: string,
    onChangeText: (value: string) => void,
    options?: {
      secure?: boolean;
      toggle?: () => void;
      visible?: boolean;
      keyboardType?: 'default' | 'email-address';
      autoCapitalize?: 'none' | 'words';
    }
  ) => (
    <View style={styles.inputGroup}>
      <Text style={styles.inputLabel}>{label}</Text>

      <View style={styles.inputWrapper}>
        <TextInput
          style={styles.input}
          placeholder={placeholder}
          placeholderTextColor={COLORS.gray}
          value={value}
          onChangeText={onChangeText}
          secureTextEntry={options?.secure && !options?.visible}
          keyboardType={options?.keyboardType || 'default'}
          autoCapitalize={options?.autoCapitalize || 'none'}
          autoCorrect={false}
        />

        {options?.toggle && (
          <TouchableOpacity
            onPress={options.toggle}
            style={styles.eyeButton}
          >
            <Text style={styles.eyeText}>
              {options.visible ? 'Скрыть' : 'Показать'}
            </Text>
          </TouchableOpacity>
        )}
      </View>
    </View>
  );

  return (
    <ScrollView
      style={styles.container}
      contentContainerStyle={styles.scrollContent}
      keyboardShouldPersistTaps="handled"
      showsVerticalScrollIndicator={false}
    >
      <TouchableOpacity style={styles.backButton} onPress={onBack}>
        <Text style={styles.backText}>← Назад</Text>
      </TouchableOpacity>

      <View style={styles.brand}>
        <View style={styles.logoCircle}>
          <Text style={styles.logoEmoji}>☕</Text>
        </View>

        <Text style={styles.logoTitle}>AROMA CRAFT</Text>
        <Text style={styles.logoSubtitle}>COFFEE & DESSERTS</Text>
      </View>

      <View style={styles.card}>
        {screen === 'login' && (
          <>
            <Text style={styles.title}>С возвращением!</Text>
            <Text style={styles.subtitle}>
              Войдите в свой аккаунт
            </Text>

            {renderInput(
              'Электронная почта',
              'example@gmail.com',
              email,
              setEmail,
              {
                keyboardType: 'email-address',
              }
            )}

            {renderInput(
              'Пароль',
              'Введите пароль',
              password,
              setPassword,
              {
                secure: true,
                visible: showPassword,
                toggle: () => setShowPassword(!showPassword),
              }
            )}

            <TouchableOpacity
              style={styles.forgotButton}
              onPress={() => setScreen('forgot')}
            >
              <Text style={styles.forgotText}>
                Забыли пароль?
              </Text>
            </TouchableOpacity>

            <TouchableOpacity
              style={styles.mainButton}
              onPress={handleLogin}
            >
              <Text style={styles.mainButtonText}>Войти</Text>
            </TouchableOpacity>

            <View style={styles.switchRow}>
              <Text style={styles.switchText}>
                Нет аккаунта?
              </Text>

              <TouchableOpacity
                onPress={() => setScreen('register')}
              >
                <Text style={styles.switchLink}>
                  {' '}Зарегистрироваться
                </Text>
              </TouchableOpacity>
            </View>
          </>
        )}

        {screen === 'register' && (
          <>
            <Text style={styles.title}>Создать аккаунт</Text>
            <Text style={styles.subtitle}>
              Присоединяйтесь к AROMA CRAFT
            </Text>

            {renderInput(
              'Ваше имя',
              'Введите имя',
              name,
              setName,
              {
                autoCapitalize: 'words',
              }
            )}

            {renderInput(
              'Электронная почта',
              'example@gmail.com',
              email,
              setEmail,
              {
                keyboardType: 'email-address',
              }
            )}

            {renderInput(
              'Пароль',
              'Минимум 6 символов',
              password,
              setPassword,
              {
                secure: true,
                visible: showPassword,
                toggle: () => setShowPassword(!showPassword),
              }
            )}

            {renderInput(
              'Повторите пароль',
              'Введите пароль ещё раз',
              confirmPassword,
              setConfirmPassword,
              {
                secure: true,
                visible: showConfirmPassword,
                toggle: () =>
                  setShowConfirmPassword(!showConfirmPassword),
              }
            )}

            <TouchableOpacity
              style={styles.mainButton}
              onPress={handleRegister}
            >
              <Text style={styles.mainButtonText}>
                Зарегистрироваться
              </Text>
            </TouchableOpacity>

            <View style={styles.switchRow}>
              <Text style={styles.switchText}>
                Уже есть аккаунт?
              </Text>

              <TouchableOpacity
                onPress={() => setScreen('login')}
              >
                <Text style={styles.switchLink}> Войти</Text>
              </TouchableOpacity>
            </View>
          </>
        )}

        {screen === 'forgot' && (
          <>
            <Text style={styles.title}>Восстановление пароля</Text>
            <Text style={styles.subtitle}>
              Укажите email, который использовали при регистрации.
            </Text>

            {renderInput(
              'Электронная почта',
              'example@gmail.com',
              email,
              setEmail,
              {
                keyboardType: 'email-address',
              }
            )}

            <TouchableOpacity
              style={styles.mainButton}
              onPress={handleForgot}
            >
              <Text style={styles.mainButtonText}>
                Восстановить пароль
              </Text>
            </TouchableOpacity>

            <TouchableOpacity
              style={styles.returnButton}
              onPress={() => setScreen('login')}
            >
              <Text style={styles.returnText}>
                ← Вернуться ко входу
              </Text>
            </TouchableOpacity>
          </>
        )}
      </View>

      <Text style={styles.footer}>
        AROMA CRAFT © 2026
      </Text>
    </ScrollView>
  );
}
